{{- /* The address is built from request headers and printed as text. Refuse
       anything that is not a plain scheme and host name. */ -}}
{{- $proto := default (placeholder "http.request.scheme") (.Req.Header.Get "X-Forwarded-Proto") -}}
{{- if or (not (has $proto (list "http" "https"))) (not (regexMatch "^[A-Za-z0-9.-]+(:[0-9]+)?$" .Req.Host)) }}{{ httpError 400 }}{{ end -}}
{{- $base := printf "%s://%s" $proto .Req.Host -}}
User-agent: *
{{- if eq (env "TEHAS_INDEXING") "1" }}
Disallow: /live/
Disallow: /health

Sitemap: {{ $base }}/sitemap.xml

# For agents: {{ $base }}/llms.txt
{{- else }}
Disallow: /
{{- end }}
